  uv add "fastapi>=0.141.1" "pydantic>=2.14.0" "typing-extensions>=4.16"

  uv add uvicorn && uv run uvicorn --app-dir 'my src' server.main:app --reload
or: uv add "fastapi[standard]" && uv run fastapi dev 'my src/server/main.py'
